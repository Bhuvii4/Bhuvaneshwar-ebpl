import pandas as pd
def load_movies(file_path):
    df = pd.read_csv(file_path)
    df.columns = df.columns.str.strip().str.lower()
    df = df.dropna(subset=['genres', 'rating'])
    return df
def filter_movies_by_genre(movies, user_genre):
    filtered = movies[movies['genres'].str.contains(user_genre, case=False)]
    filtered = filtered.sort_values(by='rating', ascending=False)
    return filtered
if __name__ == "__main__":
    movies = load_movies("movie1.csv")
    print("Enter a genre to search (e.g., Action, Comedy, Horror,etc...):")
    user_input = input(">> ").strip()
    if not user_input:
        print("Please enter a genre.")
    else:
        results = filter_movies_by_genre(movies, user_input)
        if results.empty:
            print("No movies found for that genre.")
        else:
            print(f"\n Movies in genre '{user_input}' (Sorted by Rating - Descending):\n")
            for _, row in results.iterrows():
                print(f" {row['title']} ({int(row['year'])}) | Genre: {row['genres']} | Rating: {row['rating']}")
